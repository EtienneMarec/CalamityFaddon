local trim = mw.text.trim
local inputArgs

local function getArg(key)
	local v = trim(inputArgs[key] or '')

		return nil
	else
		return v
	end
end

local function tableStartTag()
	local t
	local str =  '<table class="diagram'
	t = getArg('class')
	if t then
		str = str .. ' ' .. t
	end
	str = str .. '"'
	t = getArg('id')
	if t then
		str = str .. ' id="' .. t ..'"'
	end
	t = getArg('style') or getArg('css')
	if t then
		str = str .. ' style="' .. t ..'"'
	end
	t = getArg('summary')
	if t then
		str = str .. ' title="' .. t ..'"'
	end
	return str .. '>'
end
